# Dynamic build numbers custom commands
# Re-runs the increment script whenever a dependency's mtime changes, but the
# script itself only bumps BUILD_OS when the sources' content hash actually
# differs from the last recorded one (see cmake/IncrementBuildNumber.cmake).
get_filename_component(CMD64_ENTRY_ABS "${CMD64_ENTRY}" ABSOLUTE)
set(OS_HASH_SOURCES "${CMD64_ENTRY_ABS}" ${CMD64_SRCS})
list(REMOVE_DUPLICATES OS_HASH_SOURCES)
list(SORT OS_HASH_SOURCES)
set(OS_HASH_SOURCES_FILE "${CMAKE_BINARY_DIR}/build_os_sources.txt")
string(REPLACE ";" "\n" OS_HASH_SOURCES_CONTENT "${OS_HASH_SOURCES}")
file(WRITE "${OS_HASH_SOURCES_FILE}" "${OS_HASH_SOURCES_CONTENT}\n")

add_custom_command(
    OUTPUT "${CMAKE_BINARY_DIR}/build_os.inc"
    COMMAND "${CMAKE_COMMAND}"
        -DBUILD_FILE="${CMAKE_SOURCE_DIR}/src/command64/BUILD_OS"
        -DINC_FILE="${CMAKE_BINARY_DIR}/build_os.inc"
        -DDEFAULT_VERSION=2417
        -DVAR_NAME="BUILD_NUMBER"
        -DSOURCES_LIST_FILE="${OS_HASH_SOURCES_FILE}"
        -P "${CMAKE_SOURCE_DIR}/cmake/IncrementBuildNumber.cmake"
    DEPENDS ${CMD64_SRCS} "${CMD64_ENTRY_ABS}"
    COMMENT "Checking/Incrementing OS build counter"
)

# 3e. CASM Native 6502/6510 Assembler — Phase 1 application scaffold.
# Ships from its first phase so launch, relocation, cleanup, and shell-return
# behavior use the production external-app path before parser work begins.
if(Ca65_FOUND)
    # WP6: $1000 -> $2000. WP13: $2000 -> $2800. WP19: $2800 -> $2A00. WP24:
    # $2A00 -> $2B00 -- the grown CasmVmmRegistry record plus CasmVmmBuffer
    # and vmmWindowRead/Write/Replay overflowed $2A00 by 123 measured bytes;
    # $2B00 (+256) leaves 133 bytes headroom, the smallest round-hex step
    # above that minimum, matching this line's own precedent. WP27: $2B00 ->
    # $2F00 -- symbols.s (the VMM-backed symbol table/hash index) plus the
    # CASM_VMM_BUFFER_SIZE 32->64 growth and the diagnostics.s message-table
    # fix for $28-$2F overflowed $2B00 by 848 measured bytes; $2F00 (+1024)
    # leaves 176 bytes headroom, the smallest round-page step above that
    # minimum, matching this line's own precedent. WP28: $2F00 -> $3000 --
    # CasmParserStmt's growth to 7 bytes, the new CasmLabelName/
    # CasmLabelNameLen buffer, and CasmPassMode overflowed $2F00 by 23
    # measured bytes; $3000 (+256) leaves 233 bytes headroom, the smallest
    # round-page step above that minimum, matching this line's own precedent.
    # WP40: $3500 -> $3600 -- reloc.s (the VMM-backed relocation table) plus
    # the six new emit.s call sites overflowed $3500 by 145 measured bytes;
    # $3600 (+256) leaves 111 bytes headroom, the smallest round-page step
    # above that minimum, matching this line's own precedent.
    # WP41: $3600 -> $3700 -- relocFinalize (the R6 footer/table writer)
    # overflowed $3600 by 103 measured bytes; $3700 (+256) leaves 153 bytes
    # headroom, the smallest round-page step above that minimum, matching
    # this line's own precedent.
    # WP44: $3700 -> $3A00 -- the approved 65-byte include operand state plus
    # three grammar diagnostics overflowed $3700 by 16 measured bytes before
    # scanner/dispatch code. The reviewed +768-byte envelope covers the full
    # quoted-operand work package without approaching the user-space ceiling.
    # WP45: $3A00 -> $3E00 -- the new standalone include.s catalog module
    # (metadata store, device resolution, case-fold compare, record I/O)
    # measured a 694-byte overflow at $3A00; the user approved a reviewed
    # +1024-byte envelope, leaving 330 bytes headroom.
    # WP46: $3E00 -> $4000 -- the new source.s frame stack (16-level
    # push/pop/cycle-detection arrays plus the rewired sourceRefill)
    # measured a 221-byte overflow at $3E00; the user approved growing to
    # exactly what's needed, leaving 292 bytes headroom.
    add_ca65_app(casm "${CASM_ENTRY}" CASM_SRCS 1000 "4000")
else()
    message(FATAL_ERROR "ca65/ld65 not found on PATH — required for CASM target")
endif()
set(CASM_TARGET casm)

set(SYNC_FILES
    api-reference.md
    debug-test-plan.md
    pet-sci-api.md
    programmers-reference.md
    user-manual.md
    vmm-api.md
)

set(SYNC_OUTPUTS "")
foreach(FILE ${SYNC_FILES})
    add_custom_command(
        OUTPUT "${CMAKE_SOURCE_DIR}/docs/${FILE}"
        COMMAND "${CMAKE_COMMAND}" -E copy
            "${CMAKE_SOURCE_DIR}/wiki/${FILE}"
            "${CMAKE_SOURCE_DIR}/docs/${FILE}"
        DEPENDS "${CMAKE_SOURCE_DIR}/wiki/${FILE}"
        COMMENT "Syncing ${FILE} from wiki/ to docs/"
    )
    list(APPEND SYNC_OUTPUTS "${CMAKE_SOURCE_DIR}/docs/${FILE}")
endforeach()

add_custom_target(sync_docs ALL DEPENDS ${SYNC_OUTPUTS})

# CASM WP14 trusted reference PRGs. Reviewed hex manifests under
# tests/fixtures/casm/*.ref.hex are transcribed to binary by
# scripts/hex_manifest_to_bin.py (which contains no 6502 knowledge, so it cannot
# reproduce a CASM opcode-table defect) and placed on test.d64 as PRGs, so the
# native COMP utility can byte-for-byte compare CASM output against them. See
# brain/plans/2026-07-20-casm-phase4-wp14-orchestration-binary-validation.md.
set(CASM_REF_DIR "${CMAKE_BINARY_DIR}/casm_refs")
set(CASM_REF_MANIFEST_DIR "${CMAKE_SOURCE_DIR}/tests/fixtures/casm")

    p1fwd1 p1back1 p1size1 brfwd1 brback1 casmcase1 casmmaxid1
    casmmf1 casmmf2 casmmf3 casmbig1
    casmorg1 casmorgexpl1 casmnoorg1
    casmordhaz1
    casmrelop1 casmrelop2
    casmreloc1)
set(CASM_REF_BINS "")
foreach(REF_NAME IN LISTS CASM_REF_NAMES)
    set(REF_MANIFEST "${CASM_REF_MANIFEST_DIR}/${REF_NAME}.ref.hex")
    set(REF_BIN "${CASM_REF_DIR}/${REF_NAME}.ref")
    add_custom_command(
        OUTPUT "${REF_BIN}"
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${CASM_REF_DIR}"
        COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_SOURCE_DIR}/scripts/hex_manifest_to_bin.py"
            "${REF_MANIFEST}" "${REF_BIN}"
        DEPENDS "${REF_MANIFEST}"
            "${CMAKE_SOURCE_DIR}/scripts/hex_manifest_to_bin.py"
        COMMENT "Generating CASM trusted reference ${REF_NAME}.ref from manifest"
        VERBATIM
    )
    list(APPEND CASM_REF_BINS "${REF_BIN}")
endforeach()

# Large-source CASM fixtures whose raw .seq text does not fit on test.d64
# alongside every other CASM/OS fixture, so they get their own small,
# dedicated disk image instead:
#   - casmmfovf1.seq/casmmfovf2.seq (WP34, 40000/30000 bytes): prove the
#     real 65535-byte combined multi-file source cap, which architecturally
#     cannot be exercised with less content.
#   - casmbiga.seq/casmbigb.seq (WP36, 12011/12000 bytes): prove a large,
#     under-cap, two-file source assembles successfully -- closes the
#     master plan's Phase 7 gate-text gap ("large ... inputs assemble
#     successfully"), never proven at real scale by any prior fixture.
# Both pairs are generated by the same casm_test_fixtures script as every
# other CASM fixture (see cmake/GenerateCasmTestFixtures.cmake); neither is
# listed in CASM_TEST_FIXTURES itself since that list also drives test.d64's
# packaging loop below. casmbig1 needs COMP verification (unlike the
# overflow pair, which only needs a diagnostic to fire), so this disk also
# carries comp.prg and the casmbig1.ref trusted-reference PRG -- excluded
# from the generic CASM_REF_NAMES -> test.d64 loop below since its matching
# .seq inputs live only on this disk, not test.d64.
#
# test_l15release also lives here, despite having nothing to do with CASM:
# test.d64 sits at the 1541 directory track's hard 144-entry ceiling (not a
# byte-space limit -- 139 blocks were still free), so adding it there would
# have bumped an unrelated fixture off. This disk is mechanically just "a
# D64 with spare directory entries," not CASM-exclusive in practice, and
# reusing it was smaller-blast-radius than standing up a new disk image
# for one 4-block test PRG. See brain/plans/label-l15-cache-release.md.
add_c64_disk_image(casm_overflow_test_d64
    OUTPUT_FILE "${CMAKE_BINARY_DIR}/casm_overflow_test.d64"
    DISK_LABEL "${DISK_NAME}"
    DISK_ID "${DISK_ID}"
    PRGS casm comp test_l15release test_casm_include test_casm_catalog test_casm_frame
)
add_dependencies(casm_overflow_test_d64 casm_test_fixtures)
add_dependencies(casm_overflow_test_d64 casm_reference_fixtures)

# CASM source fixtures carry an explicit ".s" suffix in their on-disk
# PETSCII name (e.g. "casmhello.s"), distinct from the bare-named PRG
# binaries elsewhere on this disk (and from the ".ref" trusted-reference
# PRGs below) -- a directory listing's TYPE column is easy to miss, so the
# name itself should make "this is source, not a program" obvious. Applies
# only to real CASM assembly source text; the generic testseq/edlin*
# SEQ fixtures below are plain text content, not source code, and keep
# their existing bare names.
foreach(CASM_FIXTURE IN LISTS CASM_TEST_FIXTURES)
    get_filename_component(CASM_FIXTURE_NAME "${CASM_FIXTURE}" NAME_WE)
    set(CASM_FIXTURE_DISK_NAME "${CASM_FIXTURE_NAME}.s")
    add_custom_command(
        TARGET test_image_d64 POST_BUILD
        COMMAND "${CC1541_EXECUTABLE}" -f "${CASM_FIXTURE_DISK_NAME}" -T SEQ
            -w "${CASM_FIXTURE}" "${CMAKE_BINARY_DIR}/test.d64"
        COMMENT "Appending ${CASM_FIXTURE_DISK_NAME} SEQ fixture to test.d64"
        VERBATIM
    )
endforeach()

# 6. Release Target
# Gather built binary paths to pass to release script
get_target_property(CMD64_PATH command64 C64_PRG_PATH)
get_target_property(DEBUG_PATH debug C64_PRG_PATH)
set(PRG_PATHS_LIST "${CMD64_PATH};${DEBUG_PATH}")

foreach(C_TARGET ${C_TARGETS})
    get_target_property(C_PATH ${C_TARGET} C64_PRG_PATH)
    list(APPEND PRG_PATHS_LIST "${C_PATH}")
endforeach()

string(REPLACE ";" "," PRG_PATHS_STR "${PRG_PATHS_LIST}")

add_custom_target(release
    COMMAND "${CMAKE_COMMAND}"
        -DVERSION="${OS_VERSION}"
        -DSOURCE_DIR="${CMAKE_SOURCE_DIR}"
        -DBINARY_DIR="${CMAKE_BINARY_DIR}"
        -DRELEASE_DIR="${CMAKE_SOURCE_DIR}/release"
        -DPRG_PATHS="${PRG_PATHS_STR}"
        -P "${CMAKE_SOURCE_DIR}/cmake/PackRelease.cmake"
    DEPENDS image_d64 test_image_d64
    COMMENT "Packaging command64 OS v${OS_VERSION} release archives"
)
