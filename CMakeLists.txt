# Origin for user executable programs (default $3800).
#
# This has been raised whenever resident OS segments outgrew the space below it
# ($2000 -> $2200 -> $2600 -> $2C00 -> $3200 -> $3400 -> $3800). The move to
# $3800 accommodates the DOS_GET_SYSTEM_INFO ($5C) / DOS_GET_APP_INFO ($5D)
# handler bodies, which pushed the OS image to $3538.
#
# NOTE: this is deliberately NOT the same thing as CASM's emission origin
# (CASM_DEFAULT_ORIGIN, $3400, in src/external/casm/common.inc). CASM emits
# relocatable R6 output whose footer records $3400 as the relocation base; the
# OS loader relocates it to wherever it is actually loaded, so the two values
# are free to differ. Anything compared byte-for-byte against native CASM
# output must be linked at CASM's origin -- see add_ca65_app's BASE_HEX.
set(USER_PROG_START_HEX "3800" CACHE STRING "Hex origin for user programs")
set(USER_PROG_START_HEX_NEXT "3900" CACHE STRING "Hex origin for next page relocation build")
set(USER_PROG_START_ADDR "$${USER_PROG_START_HEX}")
set(USER_PROG_START_ADDR_NEXT "$${USER_PROG_START_HEX_NEXT}")
file(WRITE "${CMAKE_BINARY_DIR}/build_config_default/build_config.inc" "// Generated by CMake\n.const UserProgStart = ${USER_PROG_START_ADDR}\n")

# Read version file. Now that VERSION feeds a runtime-visible OS field
# (DOS_GET_SYSTEM_INFO's OsMajor/OsMinor/OsPatch/OsStage, see below) rather
# than only release-packaging filenames, re-run configure whenever it changes.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/VERSION")
if(EXISTS "${CMAKE_SOURCE_DIR}/VERSION")
    file(READ "${CMAKE_SOURCE_DIR}/VERSION" OS_VERSION)
    string(STRIP "${OS_VERSION}" OS_VERSION)
else()
    set(OS_VERSION "${PROJECT_VERSION}")
endif()

# Parse OS_VERSION (MAJOR.MINOR.PATCH[-dev]) into DOS_GET_SYSTEM_INFO's
# OsMajor/OsMinor/OsPatch/OsStage fields. WP6 Amendment to the WP1 record
# contract (brain/plans/2026-07-26-casm-dash-wp1-api-contract-freeze.md
# section 7): these were previously hardcoded immediates in ahGetSystemInfo,
# disconnected from this VERSION file.
string(REGEX MATCH "^([0-9]+)\\.([0-9]+)\\.([0-9]+)(-dev)?$" OS_VERSION_MATCH "${OS_VERSION}")
if(NOT OS_VERSION_MATCH)
    message(FATAL_ERROR "VERSION file content '${OS_VERSION}' does not match MAJOR.MINOR.PATCH[-dev]; required to populate DOS_GET_SYSTEM_INFO's OsMajor/OsMinor/OsPatch/OsStage")
endif()
set(OS_VERSION_MAJOR "${CMAKE_MATCH_1}")
set(OS_VERSION_MINOR "${CMAKE_MATCH_2}")
set(OS_VERSION_PATCH "${CMAKE_MATCH_3}")
if(CMAKE_MATCH_4)
    set(OS_VERSION_STAGE "1")
else()
    set(OS_VERSION_STAGE "0")
endif()
foreach(OS_VERSION_COMPONENT_NAME OS_VERSION_MAJOR OS_VERSION_MINOR OS_VERSION_PATCH)
    if(${OS_VERSION_COMPONENT_NAME} GREATER 255)
        message(FATAL_ERROR "VERSION component ${${OS_VERSION_COMPONENT_NAME}} exceeds byte range (0-255) for DOS_GET_SYSTEM_INFO")
    endif()
endforeach()

# Kick dialect: appended to the same generated build_config.inc UserProgStart
# already uses (see above). Ca65 dialect: separate generated file, consumed
# only by tests/src/api/api.s (the sole ca65 source needing live version

file(APPEND "${CMAKE_BINARY_DIR}/build_config_default/build_config.inc"

    ".const OsVersionMinor = ${OS_VERSION_MINOR}\n"
    ".const OsVersionPatch = ${OS_VERSION_PATCH}\n"

    "OsVersionMinor = ${OS_VERSION_MINOR}\n"
    "OsVersionPatch = ${OS_VERSION_PATCH}\n"
    "OsVersionStage = ${OS_VERSION_STAGE}\n")

# Dynamic build numbers custom commands
# Re-runs the increment script whenever a dependency's mtime changes, but the
# script itself only bumps BUILD_OS when the sources' content hash actually
# differs from the last recorded one (see cmake/IncrementBuildNumber.cmake).

list(REMOVE_DUPLICATES OS_HASH_SOURCES)
list(SORT OS_HASH_SOURCES)
set(OS_HASH_SOURCES_FILE "${CMAKE_BINARY_DIR}/build_os_sources.txt")
string(REPLACE ";" "\n" OS_HASH_SOURCES_CONTENT "${OS_HASH_SOURCES}")
file(WRITE "${OS_HASH_SOURCES_FILE}" "${OS_HASH_SOURCES_CONTENT}\n")

add_custom_command(
    OUTPUT "${CMAKE_BINARY_DIR}/build_os.inc"
    COMMAND "${CMAKE_COMMAND}"
        -DBUILD_FILE="${CMAKE_SOURCE_DIR}/src/command64/BUILD_OS"
        -DINC_FILE="${CMAKE_BINARY_DIR}/build_os.inc"
        -DDEFAULT_VERSION=2417
        -DVAR_NAME="BUILD_NUMBER"
        -DSOURCES_LIST_FILE="${OS_HASH_SOURCES_FILE}"
        -P "${CMAKE_SOURCE_DIR}/cmake/IncrementBuildNumber.cmake"
    DEPENDS ${CMD64_SRCS} "${CMD64_ENTRY_ABS}"
    COMMENT "Checking/Incrementing OS build counter"
)

    # WP6: $1000 -> $2000. WP13: $2000 -> $2800. WP19: $2800 -> $2A00. WP24:
    # $2A00 -> $2B00 -- the grown CasmVmmRegistry record plus CasmVmmBuffer
    # and vmmWindowRead/Write/Replay overflowed $2A00 by 123 measured bytes;
    # $2B00 (+256) leaves 133 bytes headroom, the smallest round-hex step
    # above that minimum, matching this line's own precedent. WP27: $2B00 ->
    # $2F00 -- symbols.s (the VMM-backed symbol table/hash index) plus the
    # CASM_VMM_BUFFER_SIZE 32->64 growth and the diagnostics.s message-table
    # fix for $28-$2F overflowed $2B00 by 848 measured bytes; $2F00 (+1024)
    # leaves 176 bytes headroom, the smallest round-page step above that
    # minimum, matching this line's own precedent. WP28: $2F00 -> $3000 --
    # CasmParserStmt's growth to 7 bytes, the new CasmLabelName/
    # CasmLabelNameLen buffer, and CasmPassMode overflowed $2F00 by 23
    # measured bytes; $3000 (+256) leaves 233 bytes headroom, the smallest
    # round-page step above that minimum, matching this line's own precedent.
    # WP40: $3500 -> $3600 -- reloc.s (the VMM-backed relocation table) plus
    # the six new emit.s call sites overflowed $3500 by 145 measured bytes;
    # $3600 (+256) leaves 111 bytes headroom, the smallest round-page step
    # above that minimum, matching this line's own precedent.
    # WP41: $3600 -> $3700 -- relocFinalize (the R6 footer/table writer)
    # overflowed $3600 by 103 measured bytes; $3700 (+256) leaves 153 bytes
    # headroom, the smallest round-page step above that minimum, matching
    # this line's own precedent.

    # three grammar diagnostics overflowed $3700 by 16 measured bytes before
    # scanner/dispatch code. The reviewed +768-byte envelope covers the full
    # quoted-operand work package without approaching the user-space ceiling.
    # WP45: $3A00 -> $3E00 -- the new standalone include.s catalog module
    # (metadata store, device resolution, case-fold compare, record I/O)
    # measured a 694-byte overflow at $3A00; the user approved a reviewed
    # +1024-byte envelope, leaving 330 bytes headroom.
    # WP46: $3E00 -> $4000 -- the new source.s frame stack (16-level

# DASH_REF Reference Target — host ca65/ld65 build of the very same seven
# sources, existing only to be compared byte-for-byte against native CASM
# output. It is NEVER the source of shipped bytes (see the DASH production
# target below).
#
# BASE_HEX pins it to CASM's emission origin rather than USER_PROG_START_HEX:
# CASM always emits at $3400, so a reference linked anywhere else would differ
# in every absolute operand and could never match. DASH is relocatable, so the
# pinned origin has no runtime consequence.
#
# dmain.s is the single entry point for BOTH toolchains: it pulls in the other
# six sources via native CASM's .INCLUDE, using the same uppercase operand
# spelling ("DSCR.S" etc.) that already matches the disk directory entries
# cc1541 -f writes (proven by the existing multi-file CLI, which opens these
# exact files under these exact uppercase names today). ca65, though, resolves
# .include operands as literal filesystem paths on this case-sensitive host,
# and the checked-in files are lowercase (dscr.s). Rather than duplicate the
# include order a second time in dash_wrapper.s (which is exactly the
# hand-synced-in-two-places risk this restructuring removes), generate
# uppercase symlinks at configure time and hand ca65 an extra -I directory so
# the identical "DSCR.S"-spelled operand resolves for both toolchains without
# renaming any checked-in source file.
if(Ca65_FOUND)
    set(DASH_REF_SHIM_DIR "${CMAKE_BINARY_DIR}/dash_ref_includes")
    file(MAKE_DIRECTORY "${DASH_REF_SHIM_DIR}")
    foreach(DASH_SUB dscr.s dfmt.s dsys.s dapp.s dvmm.s ddata.s)
        string(TOUPPER "${DASH_SUB}" DASH_SUB_UPPER)
        file(CREATE_LINK
            "${CMAKE_SOURCE_DIR}/src/external/dash/${DASH_SUB}"
            "${DASH_REF_SHIM_DIR}/${DASH_SUB_UPPER}"
            SYMBOLIC COPY_ON_ERROR
        )
    endforeach()
    add_ca65_app(dash_ref "${DASH_ENTRY}" DASH_SRCS 1000 "1000" BASE_HEX "3400"
        EXTRA_INCLUDE_DIRS "${DASH_REF_SHIM_DIR}")
else()
    message(FATAL_ERROR "ca65/ld65 not found on PATH — required for DASH_REF target")
endif()

# DASH Production Target — generated from the reviewed hex manifest.
#
# PROVENANCE: the manifest records bytes produced by the NATIVE CASM assembler
# running on the C64 against the seven sources in src/external/dash/, packaged
# as SEQ files on command64_casm_utils.d64 (see that image target below). The
# dash_ref ca65 target is an INDEPENDENT cross-check only -- it must never be
# the source of manifest bytes. Regenerating the manifest is a deliberate,
# reviewed act (scripts/build_dash_manifest.py), never a build step, so a
# source edit can never silently change what ships.
#
# STALE-ARTIFACT PROTECTION (WP9): the manifest records a source_sha256 for
# each of the seven sources at regeneration time; --source-dir here makes
# hex_manifest_to_bin.py recompute those hashes from the real files and hard-
# fail on any mismatch (or on a manifest with no recorded hashes at all,
# e.g. one predating this check). This is a deliberate, expected build
# failure right now: dash.ref.hex is still the WP4 placeholder and has never
# been regenerated for WP5-WP8's sources, so it correctly refuses to ship a
# stub as if it were the current dashboard. Building `dash` (and therefore
# `image_d64`, which now packages it) will fail until a real native-CASM run
# is reviewed and transcribed via build_dash_manifest.py.
set(DASH_MANIFEST "${CMAKE_SOURCE_DIR}/src/external/dash/dash.ref.hex")
set(DASH_SOURCE_DIR "${CMAKE_SOURCE_DIR}/src/external/dash")
set(DASH_BIN "${CMAKE_BINARY_DIR}/dash.prg")
# Same seven sources as DASH_SEQ_SOURCES below (defined later in this file,
# alongside command64_casm_utils_d64) -- listed again here, not referenced,
# since CMake processes this file top-to-bottom and that variable does not
# exist yet at this point.
set(DASH_SOURCE_ABS_EARLY "")
foreach(DASH_SRC_NAME dmain.s dscr.s dfmt.s dsys.s dapp.s dvmm.s ddata.s)
    list(APPEND DASH_SOURCE_ABS_EARLY "${DASH_SOURCE_DIR}/${DASH_SRC_NAME}")
endforeach()
add_custom_command(
    OUTPUT "${DASH_BIN}"
    COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_SOURCE_DIR}/scripts/hex_manifest_to_bin.py"
        "${DASH_MANIFEST}" "${DASH_BIN}"
        "--source-dir" "${DASH_SOURCE_DIR}"
    DEPENDS "${DASH_MANIFEST}"
        "${CMAKE_SOURCE_DIR}/scripts/hex_manifest_to_bin.py"
        ${DASH_SOURCE_ABS_EARLY}
    COMMENT "Generating DASH production binary from manifest"
    VERBATIM
)
add_custom_target(dash ALL DEPENDS "${DASH_BIN}")
set_target_properties(dash PROPERTIES C64_PRG_PATH "${DASH_BIN}")

# DASH's seven sources, as SEQ files, in the authoritative assembly order.
# That order is not cosmetic: CASM emits one linear stream in command-line
# order, so it -- and dash_wrapper.s's matching .include order -- is what makes
# the native and ca65 layouts comparable. ddata.s must stay last so the data
# block follows all code.
#
# Written with cc1541 -w (verbatim host bytes), so the on-disk source is the
# same ASCII the repo and ca65 see. The lowercase -f spelling matches the
# banner.s precedent already proven to open from native CASM (see
# project-cc1541-filename-case for why -f case is load-bearing).
set(DASH_SEQ_SOURCES dmain.s dscr.s dfmt.s dsys.s dapp.s dvmm.s ddata.s)

# Gate the packaging on a CASM-lexability check of the exact bytes about to be
# written. cc1541 -w does no character-set translation, so any ASCII lowercase
# here reaches the C64 as PETSCII graphics codes and CASM rejects the source --
# a failure that shows up on the C64 looking like a program or assembler
# problem, not an encoding one. Note dash_wrapper.s is deliberately absent: it
# is ca65-only, never packaged, and free to use lowercase and .segment.
set(DASH_SEQ_ABS "")
foreach(DASH_SEQ IN LISTS DASH_SEQ_SOURCES)
    list(APPEND DASH_SEQ_ABS "${CMAKE_SOURCE_DIR}/src/external/dash/${DASH_SEQ}")
endforeach()
add_custom_command(
    TARGET command64_casm_utils_d64 PRE_BUILD
    COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_SOURCE_DIR}/scripts/check_casm_source_bytes.py"
        ${DASH_SEQ_ABS} "${CMAKE_SOURCE_DIR}/src/external/banner/banner.s"
    COMMENT "Verifying native-CASM sources are uppercase ASCII"
    VERBATIM
)

foreach(DASH_SEQ IN LISTS DASH_SEQ_SOURCES)
    add_custom_command(
        TARGET command64_casm_utils_d64 POST_BUILD
        COMMAND "${CC1541_EXECUTABLE}" -f "${DASH_SEQ}" -T SEQ
            -w "${CMAKE_SOURCE_DIR}/src/external/dash/${DASH_SEQ}"
            "${CMAKE_BINARY_DIR}/command64_casm_utils.d64"
        COMMENT "Appending DASH source ${DASH_SEQ} to command64_casm_utils.d64"
        VERBATIM
    )
endforeach()

# The ca65 reference build, shipped as DASH.REF so the native run can be
# verified on the C64 itself with "COMP DASH.PRG DASH.REF".
#
# This is a genuinely non-circular check, the same principle as CASM's

# WP47 end-to-end .INCLUDE fixtures live on their own disk image
# (casm_include_test_d64, defined below) rather than joining
# casm_overflow_test.d64: that disk is down to ~10 free blocks, almost
# entirely consumed by WP34's 277-block combined-cap overflow pair, and this
# work package's walkthrough assembles eight sources that each WRITE an
# output PRG back to disk. A verification disk with no room to write the
# thing being verified -- or to re-run after a fix -- is a trap, and the
# established precedent for that squeeze is a new image, which is how
# casm_overflow_test.d64 itself came to exist.

# Append the WP14 trusted reference PRGs to test.d64 (PRG type, lowercase -f so
# they store as unshifted PETSCII matching what the user types on the default
# uppercase charset -- same convention as the SEQ fixtures above). casmbig1
# is excluded: its matching casmbiga.s/casmbigb.s inputs live only on
# casm_overflow_test_d64 (disk-space reasons above), so its reference PRG is
# appended there instead, not here. WP60 Increment 5's casmopall is excluded
# for the same reason: its matching casmopall.s source lives only on the new
# dedicated casm_opcode_test_d64 (see that target below), not test.d64.
foreach(REF_NAME IN LISTS CASM_REF_NAMES)
    if(NOT REF_NAME STREQUAL "casmbig1" AND NOT REF_NAME STREQUAL "casmopall")
        add_custom_command(
            TARGET test_image_d64 POST_BUILD
            COMMAND "${CC1541_EXECUTABLE}" -f "${REF_NAME}.ref" -T PRG
                -w "${CASM_REF_DIR}/${REF_NAME}.ref" "${CMAKE_BINARY_DIR}/test.d64"
            COMMENT "Appending CASM reference ${REF_NAME}.ref (PRG) to test.d64"
            VERBATIM
        )
    endif()
endforeach()

# CASM source fixtures carry an explicit ".s" suffix in their on-disk
# PETSCII name (e.g. "casmhello.s"), distinct from the bare-named PRG
# binaries elsewhere on this disk (and from the ".ref" trusted-reference
# PRGs below) -- a directory listing's TYPE column is easy to miss, so the
# name itself should make "this is source, not a program" obvious. Applies
# only to real CASM assembly source text; the generic testseq/edlin*
# SEQ fixtures below are plain text content, not source code, and keep
# their existing bare names.

    set(CASM_FIXTURE_DISK_NAME "${CASM_FIXTURE_NAME}.s")

        COMMAND "${CC1541_EXECUTABLE}" -f "${CASM_FIXTURE_DISK_NAME}" -T SEQ
            -w "${CASM_FIXTURE}" "${CMAKE_BINARY_DIR}/test.d64"
        COMMENT "Appending ${CASM_FIXTURE_DISK_NAME} SEQ fixture to test.d64"
        VERBATIM
    )
endforeach()
